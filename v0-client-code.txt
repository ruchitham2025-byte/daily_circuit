import { useCallback, useEffect, useState } from 'react';
import type {
  InitResponse,
  IncrementResponse,
  DecrementResponse,
} from '../../shared/api';

interface CounterState {
  count: number;
  username: string | null;
  loading: boolean;
}

export const useCounter = () => {
  const [state, setState] = useState<CounterState>({
    count: 0,
    username: null,
    loading: true,
  });
  const [postId, setPostId] = useState<string | null>(null);

  // fetch initial data
  useEffect(() => {
    const init = async () => {
      try {
        const res = await fetch('/api/init');
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const data: InitResponse = await res.json();
        if (data.type !== 'init') throw new Error('Unexpected response');
        setState({
          count: data.count,
          username: data.username,
          loading: false,
        });
        setPostId(data.postId);
      } catch (err) {
        console.error('Failed to init counter', err);
        setState((prev) => ({ ...prev, loading: false }));
      }
    };
    void init();
  }, []);

  const update = useCallback(
    async (action: 'increment' | 'decrement') => {
      if (!postId) {
        console.error('No postId â€“ cannot update counter');
        return;
      }
      try {
        const res = await fetch(`/api/${action}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({}),
        });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const data: IncrementResponse | DecrementResponse = await res.json();
        setState((prev) => ({ ...prev, count: data.count }));
      } catch (err) {
        console.error(`Failed to ${action}`, err);
      }
    },
    [postId]
  );

  const increment = useCallback(() => update('increment'), [update]);
  const decrement = useCallback(() => update('decrement'), [update]);

  return {
    ...state,
    increment,
    decrement,
  } as const;
};

export const playSound = (sound: Sound): void => {
  if (!soundEnabled || typeof window === 'undefined') return;

  try {
    const AudioContextConstructor = window.AudioContext;
    if (!AudioContextConstructor) return;

    audioContext ??= new AudioContextConstructor();
    const context = audioContext;
    if (context.state === 'suspended') {
      void context.resume().catch((error: unknown) => {
        console.error('Unable to start puzzle audio:', error);
      });
    }

    const now = context.currentTime;
    const oscillator = context.createOscillator();
    const gain = context.createGain();
    const soundSettings: Record<Sound, { frequency: number; duration: number; type: OscillatorType }> = {
      place: { frequency: 420, duration: 0.07, type: 'sine' },
      rotate: { frequency: 620, duration: 0.09, type: 'triangle' },
      receiver: { frequency: 880, duration: 0.13, type: 'sine' },
      hazard: { frequency: 145, duration: 0.22, type: 'sawtooth' },
      hint: { frequency: 720, duration: 0.1, type: 'triangle' },
      win: { frequency: 523.25, duration: 0.48, type: 'triangle' },
      laser: { frequency: 880, duration: 0.16, type: 'sine' },
    };
    const settings = soundSettings[sound];

    oscillator.type = settings.type;
    oscillator.frequency.setValueAtTime(settings.frequency, now);
    gain.gain.setValueAtTime(0.0001, now);
    gain.gain.exponentialRampToValueAtTime(0.12, now + 0.012);
    gain.gain.exponentialRampToValueAtTime(0.0001, now + settings.duration);
    oscillator.connect(gain);
    gain.connect(context.destination);
    if (sound === 'win') {
      const notes = [523.25, 659.25, 783.99, 1046.5];
      oscillator.frequency.setValueAtTime(notes[0] ?? 523.25, now);
      for (let index = 1; index < notes.length; index += 1) {
        const note = notes[index];
        if (note !== undefined) {
          oscillator.frequency.setValueAtTime(note, now + index * 0.11);
        }
      }
      oscillator.start(now);
      oscillator.stop(now + settings.duration);
    } else {
      if (sound === 'receiver') {
        oscillator.frequency.exponentialRampToValueAtTime(
          settings.frequency * 1.25,
          now + settings.duration
        );
      }
      oscillator.start(now);
      oscillator.stop(now + settings.duration + 0.02);
    }
  } catch (error) {
    console.error('Unable to play puzzle sound:', error);
  }
};

export type CampaignGrade = 'S' | 'A' | 'B' | 'C';

export type CampaignProgressData = {
  unlockedThrough: number;
  bestGrades: Record<number, CampaignGrade>;

export type CampaignProgressSnapshot =
  | { status: 'loading' }
  | { status: 'error' }
  | { status: 'ready'; data: CampaignProgressData };

const loadingSnapshot: CampaignProgressSnapshot = { status: 'loading' };
let snapshot: CampaignProgressSnapshot = loadingSnapshot;
const listeners = new Set<() => void>();

const STORAGE_KEY = 'daily-circuit:campaign-progress';

const isRecord = (value: unknown): value is Record<string, unknown> =>
  typeof value === 'object' && value !== null && !Array.isArray(value);

const isGrade = (value: unknown): value is CampaignGrade =>
  value === 'S' || value === 'A' || value === 'B' || value === 'C';

const GRADE_ORDER: CampaignGrade[] = ['S', 'A', 'B', 'C'];

const notifyListeners = (): void => {
  listeners.forEach((listener) => listener());
};

const loadLocalProgress = (): CampaignProgressData => {
  try {
    if (typeof window === 'undefined') return { unlockedThrough: 0, bestGrades: {} };
    const raw = window.localStorage.getItem(STORAGE_KEY);
    if (!raw) return { unlockedThrough: 0, bestGrades: {} };
    const parsed: unknown = JSON.parse(raw);
    if (!isRecord(parsed)) return { unlockedThrough: 0, bestGrades: {} };

    const bestGrades: Record<number, CampaignGrade> = {};
    if (isRecord(parsed.bestGrades)) {
      for (const [key, value] of Object.entries(parsed.bestGrades)) {
        const id = Number(key);
        if (Number.isSafeInteger(id) && id >= 1 && id <= 30 && isGrade(value)) {
          bestGrades[id] = value;
        }
      }
    }

    let unlockedThrough = 0;
    const storedUnlocked = parsed.unlockedThrough;
    if (typeof storedUnlocked === 'number' && Number.isSafeInteger(storedUnlocked)) {
      unlockedThrough = Math.max(0, Math.min(30, storedUnlocked));
    } else {
      for (let id = 1; id <= 30; id += 1) {
        if (bestGrades[id] === undefined) break;
        unlockedThrough = id;
      }
    }

    return { unlockedThrough, bestGrades };
  } catch {
    return { unlockedThrough: 0, bestGrades: {} };
  }
};

const saveLocalProgress = (data: CampaignProgressData): void => {
  try {
    if (typeof window === 'undefined') return;
    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
  } catch {
    // ignore storage errors
  }
};

export const subscribeCampaignProgress = (listener: () => void): (() => void) => {
  listeners.add(listener);
  return () => listeners.delete(listener);
};

export const getCampaignProgress = (): CampaignProgressSnapshot => snapshot;

export const refreshCampaignProgress = async (): Promise<void> => {
  snapshot = loadingSnapshot;
  notifyListeners();

  try {
    if (import.meta.env.DEV || !navigator.onLine) throw new Error('Local session');
    const response = await fetch('/api/campaign-progress', {signal:AbortSignal.timeout(4000)});
    if (!response.ok) {
      throw new Error(`Campaign progress request failed (${response.status})`);
    }

    const payload: unknown = await response.json();
    if (
      !isRecord(payload) ||
      payload.success !== true ||
      typeof payload.unlockedThrough !== 'number' ||
      !Number.isSafeInteger(payload.unlockedThrough) ||
      payload.unlockedThrough < 0 ||
      payload.unlockedThrough > 30 ||
      !isRecord(payload.bestGrades)
    ) {
      throw new Error('Campaign progress response was invalid');
    }

    const bestGrades: Record<number, CampaignGrade> = {};
    for (const [levelKey, grade] of Object.entries(payload.bestGrades)) {
      const levelId = Number(levelKey);
      if (
        !Number.isSafeInteger(levelId) ||
        levelId < 1 ||
        levelId > 30 ||
        !isGrade(grade)
      ) {
        throw new Error('Campaign progress contained an invalid best grade');
      }
      bestGrades[levelId] = grade;
    }

    const data = {
      unlockedThrough: payload.unlockedThrough,
      bestGrades,
    };
    saveLocalProgress(data);
    snapshot = { status: 'ready', data };
  } catch {
    const local = loadLocalProgress();
    snapshot = { status: 'ready', data: local };
  }

  notifyListeners();
};

export const recordCampaignCompletion = async (
  levelId: number,
  grade: CampaignGrade
): Promise<void> => {
  try {
    if (import.meta.env.DEV || !navigator.onLine) throw new Error('Local session');
    const response = await fetch('/api/campaign-progress', {
      signal:AbortSignal.timeout(4000),
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ levelId, grade }),
    });

    if (!response.ok) {
      throw new Error(`Campaign completion could not be saved (${response.status})`);
    }

    const payload: unknown = await response.json();
    if (!isRecord(payload) || payload.success !== true) {
      throw new Error('Campaign completion was not accepted');
    }

    await refreshCampaignProgress();
    if (snapshot.status !== 'ready') {
      throw new Error('Campaign completion saved, but progress could not be refreshed');
    }
  } catch {
    const local = loadLocalProgress();
    const previousBest = local.bestGrades[levelId];
    const previousRank = previousBest ? GRADE_ORDER.indexOf(previousBest) : -1;
    const nextRank = GRADE_ORDER.indexOf(grade);
    if (previousRank < 0 || nextRank < previousRank) {
      local.bestGrades[levelId] = grade;
    }
    if (levelId > local.unlockedThrough) {
      local.unlockedThrough = levelId;
    }
    saveLocalProgress(local);
    snapshot = { status: 'ready', data: local };
    notifyListeners();
  }
};

const parseDailyState = (value: unknown): DailyStateData | undefined => {
  if (!isRecord(value) || !isRecord(value.gradeDistribution) || !isRecord(value.playerStats)) {
    return undefined;
  }

  const { gradeDistribution, playerStats } = value;
  const globalAverageCost = value.globalAverageCost;
  const averageCost = playerStats.averageCost;
  const personalBest = playerStats.personalBest;
  const personalBests = playerStats.personalBests;
  const bestTimeSeconds = playerStats.bestTimeSeconds;

  if (
    !isFiniteNumber(value.totalCompletions) ||
    !isFiniteNumber(value.playerCompletions) ||
    (globalAverageCost !== null && !isFiniteNumber(globalAverageCost)) ||
    !isFiniteNumber(gradeDistribution.sPlus) ||
    !isFiniteNumber(gradeDistribution.a) ||
    !isFiniteNumber(gradeDistribution.b) ||
    !isFiniteNumber(gradeDistribution.c) ||
    !isFiniteNumber(playerStats.streak) ||
    !isFiniteNumber(playerStats.totalPuzzles) ||
    (averageCost !== null && !isFiniteNumber(averageCost)) ||
    (personalBest !== null && !isFiniteNumber(personalBest)) ||
    !Array.isArray(personalBests) ||
    personalBests.length !== 3 ||
    (bestTimeSeconds !== null && !isFiniteNumber(bestTimeSeconds)) ||
    typeof playerStats.efficiencyRating !== 'string'
  ) {
    return undefined;
  }

  const parsedPersonalBests: { levelId: number; cost: number | null }[] = [];
  for (const best of personalBests) {
    if (
      !isRecord(best) ||
      !isFiniteNumber(best.levelId) ||
      (best.cost !== null && !isFiniteNumber(best.cost))
    ) {
      return undefined;
    }
    parsedPersonalBests.push({ levelId: best.levelId, cost: best.cost });
  }

  return {
    totalCompletions: value.totalCompletions,
    globalAverageCost,
    gradeDistribution: {
      sPlus: gradeDistribution.sPlus,
      a: gradeDistribution.a,
      b: gradeDistribution.b,
      c: gradeDistribution.c,
    },
    playerCompletions: value.playerCompletions,
    playerStats: {
      streak: playerStats.streak,
      totalPuzzles: playerStats.totalPuzzles,
      averageCost,
      personalBest,
      personalBests: parsedPersonalBests,
      bestTimeSeconds,
      efficiencyRating: playerStats.efficiencyRating,
    },
  };
};

  lesson: 'voltage-decay' | 'booster' | 'splitter' | 'hazard' | 'budget' | 'advanced';
  placements: Placement[];
  targetCost: number;
};

type Random = () => number;

const MAX_ATTEMPTS = 8;
const MIN_SIZE = 3;
const MAX_SIZE = 12;

const seededRandom = (seed: number): Random => {
  let state = seed >>> 0;
  return () => {
    state += 0x6d2b79f5;
    let value = state;
    value = Math.imul(value ^ (value >>> 15), value | 1);
    value ^= value + Math.imul(value ^ (value >>> 7), value | 61);
    return ((value ^ (value >>> 14)) >>> 0) / 4294967296;
  };
};

const shuffled = <T,>(values: readonly T[], random: Random): T[] => {
  const result = [...values];
  for (let index = result.length - 1; index > 0; index -= 1) {
    const swapIndex = Math.floor(random() * (index + 1));
    const current = result[index];
    const swap = result[swapIndex];
    if (current === undefined || swap === undefined) continue;
    result[index] = swap;
    result[swapIndex] = current;
  }
  return result;
};

const position = (size: number, row: number, column: number): number =>
  row * size + column;

const coordinate = (
  size: number,
  index: number
): { row: number; column: number } => ({
  row: Math.floor(index / size),
  column: index % size,
});

const adjacent = (size: number, index: number): number[] => {
  const { row, column } = coordinate(size, index);
  return [
    row > 0 ? position(size, row - 1, column) : -1,
    column + 1 < size ? position(size, row, column + 1) : -1,
    row + 1 < size ? position(size, row + 1, column) : -1,
    column > 0 ? position(size, row, column - 1) : -1,
  ].filter((nextIndex) => nextIndex >= 0);
};

const buildRoute = (
  size: number,
  source: number,
  target: number,
  blocked: ReadonlySet<number>,
  random: Random
): number[] | null => {
  const previous = new Map<number, number>();
  const visited = new Set<number>([source]);
  const queue = [source];

  while (queue.length > 0) {
    const current = queue.shift();
    if (current === undefined) break;
    if (current === target) break;

    for (const next of shuffled(adjacent(size, current), random)) {
      if (visited.has(next) || blocked.has(next)) continue;
      visited.add(next);
      previous.set(next, current);
      queue.push(next);
    }
  }

  if (!visited.has(target)) return null;

  const route = [target];
  let current = target;
  while (current !== source) {
    const parent = previous.get(current);
    if (parent === undefined) return null;
    route.push(parent);
    current = parent;
  }
  route.reverse();
  return route;
};

const createLessonLayout = (
  seed: number,
  size: number,
  difficulty: number,
  random: Random
): GeneratedLevel | null => {
  const center = Math.floor(size / 2);
  const source = position(size, center, center);
  const lesson =
    difficulty === 1 ? 'voltage-decay'
      : difficulty === 2 ? 'booster'
        : difficulty === 3 ? 'splitter'
          : difficulty === 4 ? 'hazard'
            : difficulty === 5 ? 'budget'
              : 'advanced';
  const needsSplitter = lesson === 'splitter' || lesson === 'budget';

  if (needsSplitter) {
    const split = position(size, center - 1, center);
    const branchLength = Math.min(center, size - 1 - center);
    const leftTarget = position(size, center - 1, center - branchLength);
    const rightTarget = position(size, center - 1, center + branchLength);
    const leftBranch = branchLength > 1
      ? [position(size, center - 1, center - 1), leftTarget]
      : [leftTarget];
    const rightBranch = branchLength > 1
      ? [position(size, center - 1, center + 1), rightTarget]
      : [rightTarget];
    const routes = [[source, split, ...leftBranch], [split, ...rightBranch]];
    const targets = [leftTarget, rightTarget];
    return createSplitLesson(
      seed,
      size,
      difficulty,
      lesson,
      routes,
      targets,
      source,
      split,
      random
    );
  }

  const target = lesson === 'booster'
    ? position(
      size,
      random() >= 0.5 ? size - 1 : 0,
      random() >= 0.5 ? size - 1 : 0
    )
    : random() >= 0.5
      ? position(size, center, random() >= 0.5 ? size - 1 : 0)
      : position(size, random() >= 0.5 ? size - 1 : 0, center);
  const idealRoute = buildRoute(size, source, target, new Set(), random);
  if (!idealRoute || idealRoute.length < 2) return null;

  const reserved = new Set(idealRoute);
  const blocked = new Set<number>();
  const hazards = new Set<number>();
  const obstacleCandidates = shuffled(
    Array.from({ length: size * size }, (_, index) => index)
      .filter((index) =>
        !reserved.has(index) &&
        adjacent(size, index).every((neighbor) => !reserved.has(neighbor))
      ),
    random
  );
  const wallCount = difficulty < 6 ? Math.min(1, Math.floor(size / 6)) :
    Math.min(2 + Math.floor((difficulty - 6) / 8), Math.floor(size));
  const hazardCount = lesson === 'hazard' ? 1 :
    difficulty >= 6 ? Math.min(1 + Math.floor((difficulty - 6) / 12), 3) : 0;
  let obstacleIndex = 0;

  for (let count = 0; count < wallCount; count += 1) {
    const cell = obstacleCandidates[obstacleIndex];
    if (cell === undefined) break;
    obstacleIndex += 1;
    blocked.add(cell);
  }
  for (let count = 0; count < hazardCount; count += 1) {
    const cell = obstacleCandidates[obstacleIndex];
    if (cell === undefined) break;
    obstacleIndex += 1;
    hazards.add(cell);
  }

  const blockedCells = new Set([...blocked, ...hazards]);
  const route = buildRoute(size, source, target, blockedCells, random);
  if (!route) return null;

  const puzzle: Puzzle = {
    size,
    source,
    targets: [],
    boosters: [],
    walls: [...blocked],
    hazards: [...hazards],
  };
  const placements = createPlacements(puzzle);
  const routeInteriors = route.slice(1, -1);
  if (lesson === 'booster' && routeInteriors.length > 0) {
    const boosterOffset = Math.min(routeInteriors.length - 1, Math.floor(routeInteriors.length / 2));
    const booster = routeInteriors[boosterOffset];
    if (booster !== undefined) puzzle.boosters = [booster];
  } else if (difficulty >= 6) {
    puzzle.boosters = routeInteriors.filter((_, offset) => offset % 4 === 3);
  }

  const reservedByCircuit = new Set<number>(route);
  if ([...hazards].some((hazard) =>
    adjacent(size, hazard).some((neighbor) => reservedByCircuit.has(neighbor))
  )) {
    return null;
  }

  for (const index of puzzle.walls) {
    const placement = placements[index];
    if (placement) placements[index] = { ...placement, fixed: 'wall' };
  }
  for (const index of puzzle.hazards) {
    const placement = placements[index];
    if (placement) placements[index] = { ...placement, fixed: 'hazard' };
  }
  for (const index of puzzle.boosters) {
    const placement = placements[index];
    if (placement) placements[index] = { ...placement, fixed: 'booster', type: 4 };
  }

  for (let pathIndex = 1; pathIndex < route.length - 1; pathIndex += 1) {
    const index = route[pathIndex];
    const previous = route[pathIndex - 1];
    const next = route[pathIndex + 1];
    if (index === undefined || previous === undefined || next === undefined) return null;
    if (puzzle.boosters.includes(index)) continue;

    const requiredDirections = [
      directionBetween(size, index, previous),
      directionBetween(size, index, next),
    ];
    const options = shuffled(
      componentOptions(puzzle, placements, index)
        .filter((option) => requiredDirections.every((direction) => option.connections.includes(direction))),
      random
    );
    const option = options[0];
    if (!option) return null;
    placements[index] = option.placement;
  }

  puzzle.targets = [{ index: target, voltage: 1 }];
  const targetPlacement = placements[target];
  if (targetPlacement) placements[target] = { ...targetPlacement, fixed: 'target', type: 4 };
  const initialEvaluation = evaluate(puzzle, placements);
  if (initialEvaluation.hazardPowered) return null;
  const targetVoltage = initialEvaluation.voltages.get(target) ?? 0;
  if (targetVoltage < 1) return null;
  puzzle.targets = [{ index: target, voltage: targetVoltage }];

  const solution = solve(puzzle);
  if (!solution) return null;
  const solvedEvaluation = evaluate(puzzle, solution);
  if (
    solvedEvaluation.hazardPowered ||
    solvedEvaluation.receivers.some((receiver) => !receiver.met)
  ) {
    return null;
  }

  return {
    ...puzzle,
    seed,
    difficulty,
    lesson,
    placements: solution,
    targetCost: solvedEvaluation.hardwareCost,
  };
};

const createPlacements = (puzzle: Puzzle): Placement[] =>
  Array.from({ length: puzzle.size * puzzle.size }, (_, index) => {
    if (index === puzzle.source) return { type: 4, rotation: 0, fixed: 'source' };
    if (puzzle.targets.some((target) => target.index === index)) {
      return { type: 4, rotation: 0, fixed: 'target' };
    }
    if (puzzle.boosters.includes(index)) {
      return { type: 4, rotation: 0, fixed: 'booster' };
    }
    if (puzzle.walls.includes(index)) return { type: 0, rotation: 0, fixed: 'wall' };
    if (puzzle.hazards.includes(index)) return { type: 4, rotation: 0, fixed: 'hazard' };
    return { type: 0, rotation: 0, fixed: 'none' };
  });

type ComponentOption = {
  placement: Placement;
  connections: readonly number[];
};

const componentOptions = (
  puzzle: Puzzle,
  placements: Placement[],
  index: number
): ComponentOption[] =>
  [1, 2, 3].flatMap((type) =>
    Array.from({ length: 4 }, (_, rotation) => {
      const candidate = placements.slice();
      candidate[index] = { type, rotation, fixed: 'none' };
      return {
        placement: candidate[index] ?? { type, rotation, fixed: 'none' },
        connections: evaluate(puzzle, candidate).connections.get(index) ?? [],
      };
    })
  );

const directionBetween = (size: number, from: number, to: number): number => {
  const fromCoordinate = coordinate(size, from);
  const toCoordinate = coordinate(size, to);
  if (toCoordinate.row < fromCoordinate.row) return 0;
  if (toCoordinate.column > fromCoordinate.column) return 1;
  if (toCoordinate.row > fromCoordinate.row) return 2;
  return 3;
};

const createSplitLesson = (
  seed: number,
  size: number,
  difficulty: number,
  lesson: GeneratedLevel['lesson'],
  routes: number[][],
  targets: number[],
  source: number,
  forcedSplitter: number | null,
  random: Random
): GeneratedLevel | null => {
  const occupied = new Set(routes.flat());
  const wallCandidates = shuffled(
    Array.from({ length: size * size }, (_, index) => index)
      .filter((index) =>
        !occupied.has(index) &&
        adjacent(size, index).every((neighbor) => !occupied.has(neighbor))
      ),
    random
  );
  const wallCount = difficulty >= 5 ? Math.min(2, Math.floor(size / 3)) : 0;
  const walls = wallCandidates.slice(0, wallCount);
  const puzzle: Puzzle = {
    size,
    source,
    targets: [],
    boosters: [],
    walls,
    hazards: [],
  };
  const placements = createPlacements(puzzle);

  const neighbors = new Map<number, Set<number>>();
  for (const route of routes) {
    for (let offset = 1; offset < route.length; offset += 1) {
      const current = route[offset];
      const previous = route[offset - 1];
      if (current === undefined || previous === undefined) return null;
      const currentNeighbors = neighbors.get(current) ?? new Set<number>();
      const previousNeighbors = neighbors.get(previous) ?? new Set<number>();
      currentNeighbors.add(previous);
      previousNeighbors.add(current);
      neighbors.set(current, currentNeighbors);
      neighbors.set(previous, previousNeighbors);
    }
  }

  const targetSet = new Set(targets);
  for (const [index, adjacentCells] of neighbors) {
    if (index === source || targetSet.has(index)) continue;
    const requiredDirections = [...adjacentCells].map((neighbor) =>
      directionBetween(size, index, neighbor)
    );
    const options = shuffled(
      componentOptions(puzzle, placements, index)
        .filter((option) =>
          requiredDirections.every((direction) => option.connections.includes(direction))
        ),
      random
    );
    const selectedOption = index === forcedSplitter
      ? options.find((option) => option.placement.type === 3)
      : options[0];
    if (!selectedOption) return null;
    placements[index] = selectedOption.placement;
  }

  for (const wall of walls) {
    const placement = placements[wall];
    if (placement) placements[wall] = { ...placement, fixed: 'wall' };
  }
  puzzle.targets = targets.map((index) => ({ index, voltage: 1 }));
  for (const target of targets) {
    const placement = placements[target];
    if (placement) placements[target] = { ...placement, fixed: 'target', type: 4 };
  }

  const initialEvaluation = evaluate(puzzle, placements);
  if (initialEvaluation.hazardPowered) return null;
  const evaluatedTargets = targets.map((index) => ({
    index,
    voltage: initialEvaluation.voltages.get(index) ?? 0,
  }));
  if (evaluatedTargets.some((target) => target.voltage < 1)) return null;
  puzzle.targets = evaluatedTargets;

  const solution = solve(puzzle);
  if (!solution) return null;
  const solvedEvaluation = evaluate(puzzle, solution);
  if (
    solvedEvaluation.hazardPowered ||
    solvedEvaluation.receivers.some((receiver) => !receiver.met)
  ) {
    return null;
  }

  return {
    ...puzzle,
    seed,
    difficulty,
    lesson,
    placements: solution,
    targetCost: solvedEvaluation.hardwareCost,
  };
};

const createDailyLayout = (
  seed: number,
  size: number,
  difficulty: number,
  random: Random
): GeneratedLevel | null => {
  const center = Math.floor(size / 2);
  const source = position(size, center, center);

  const targetCount = 2 + Math.floor(random() * 2);
  const targets: number[] = [];
  const targetPositions = new Set<number>([source]);
  let attempts = 0;
  while (targets.length < targetCount && attempts < 60) {
    attempts += 1;
    const row = Math.floor(random() * size);
    const col = Math.floor(random() * size);
    const idx = position(size, row, col);
    const minDist = Math.abs(row - center) + Math.abs(col - center);
    if (minDist >= 3 && !targetPositions.has(idx)) {
      targetPositions.add(idx);
      targets.push(idx);
    }
  }
  if (targets.length < 2) return null;

  const blocked = new Set<number>();
  const hazards = new Set<number>();
  const allReserved = new Set<number>([source, ...targets]);

  const obstacleCandidates = shuffled(
    Array.from({ length: size * size }, (_, index) => index)
      .filter((index) =>
        !allReserved.has(index) &&
        adjacent(size, index).every((neighbor) => !allReserved.has(neighbor))
      ),
    random
  );
  const wallCount = Math.min(2 + Math.floor(difficulty / 8), Math.floor(size / 2));
  const hazardCount = Math.min(1 + Math.floor(difficulty / 10), 3);
  let obstacleIndex = 0;
  for (let count = 0; count < wallCount; count += 1) {
    const cell = obstacleCandidates[obstacleIndex];
    if (cell === undefined) break;
    obstacleIndex += 1;
    blocked.add(cell);
    allReserved.add(cell);
  }
  for (let count = 0; count < hazardCount; count += 1) {
    const cell = obstacleCandidates[obstacleIndex];
    if (cell === undefined) break;
    obstacleIndex += 1;
    hazards.add(cell);
    allReserved.add(cell);
  }

  const routes: number[][] = [];
  const allRouteCells = new Set<number>([source]);
  for (const target of targets) {
    const route = buildRoute(size, source, target, new Set([...blocked, ...hazards, ...allRouteCells]), random);
    if (!route || route.length < 4) return null;
    for (const cell of route) allRouteCells.add(cell);
    routes.push(route);
  }

  if ([...hazards].some((hazard) =>
    adjacent(size, hazard).some((neighbor) => allRouteCells.has(neighbor))
  )) {
    return null;
  }

  const puzzle: Puzzle = {
    size,
    source,
    targets: [],
    boosters: [],
    walls: [...blocked],
    hazards: [...hazards],
  };
  const placements = createPlacements(puzzle);

  const allRouteInteriors = new Set<number>();
  for (const route of routes) {
    for (const cell of route.slice(1, -1)) allRouteInteriors.add(cell);
  }
  const boosterCount = Math.min(1 + Math.floor(difficulty / 12), 3);
  const interiorArray = shuffled([...allRouteInteriors], random);
  const boosters: number[] = [];
  for (let count = 0; count < Math.min(boosterCount, interiorArray.length); count += 1) {
    const cell = interiorArray[count];
    if (cell === undefined) break;
    boosters.push(cell);
  }
  puzzle.boosters = boosters;

  for (const index of puzzle.walls) {
    const placement = placements[index];
    if (placement) placements[index] = { ...placement, fixed: 'wall' };
  }
  for (const index of puzzle.hazards) {
    const placement = placements[index];
    if (placement) placements[index] = { ...placement, fixed: 'hazard' };
  }
  for (const index of puzzle.boosters) {
    const placement = placements[index];
    if (placement) placements[index] = { ...placement, fixed: 'booster', type: 4 };
  }

  const neighbors = new Map<number, Set<number>>();
  for (const route of routes) {
    for (let offset = 1; offset < route.length; offset += 1) {
      const current = route[offset];
      const previous = route[offset - 1];
      if (current === undefined || previous === undefined) return null;
      const currentNeighbors = neighbors.get(current) ?? new Set<number>();
      const previousNeighbors = neighbors.get(previous) ?? new Set<number>();
      currentNeighbors.add(previous);
      previousNeighbors.add(current);
      neighbors.set(current, currentNeighbors);
      neighbors.set(previous, previousNeighbors);
    }
  }

  const targetSet = new Set(targets);
  for (const [index, adjacentCells] of neighbors) {
    if (index === source || targetSet.has(index)) continue;
    if (puzzle.boosters.includes(index)) continue;

    const requiredDirections = [...adjacentCells].map((neighbor) =>
      directionBetween(size, index, neighbor)
    );
    const options = shuffled(
      componentOptions(puzzle, placements, index)
        .filter((option) =>
          requiredDirections.every((direction) => option.connections.includes(direction))
        ),
      random
    );
    const option = options[0];
    if (!option) return null;
    placements[index] = option.placement;
  }

  puzzle.targets = targets.map((index) => ({ index, voltage: 1 }));
  for (const target of targets) {
    const placement = placements[target];
    if (placement) placements[target] = { ...placement, fixed: 'target', type: 4 };
  }

  const initialEvaluation = evaluate(puzzle, placements);
  if (initialEvaluation.hazardPowered) return null;
  const evaluatedTargets = targets.map((index) => ({
    index,
    voltage: initialEvaluation.voltages.get(index) ?? 0,
  }));
  if (evaluatedTargets.some((target) => target.voltage < 1)) return null;
  puzzle.targets = evaluatedTargets;

  const solution = solve(puzzle);
  if (!solution) return null;
  const solvedEvaluation = evaluate(puzzle, solution);
  if (
    solvedEvaluation.hazardPowered ||
    solvedEvaluation.receivers.some((receiver) => !receiver.met)
  ) {
    return null;
  }

  const solutionParts = solution.filter(
    (placement) => placement.fixed === 'none' && placement.type > 0
  ).length;
  if (solutionParts < 7 || solvedEvaluation.hardwareCost < 85) return null;

  return {
    ...puzzle,
    seed,
    difficulty,
    lesson: 'advanced',
    placements: solution,
    targetCost: solvedEvaluation.hardwareCost,
  };
};

export const generate = (
  seed: number,
  size: number,
  difficulty: number
): GeneratedLevel | null => {
  if (!Number.isInteger(seed)) throw new Error('Seed must be an integer.');
  if (!Number.isInteger(size) || size < MIN_SIZE || size > MAX_SIZE) {
    throw new Error(`Size must be an integer from ${MIN_SIZE} to ${MAX_SIZE}.`);
  }
  if (!Number.isInteger(difficulty) || difficulty < 1 || difficulty > 30) {
    throw new Error('Difficulty must be an integer from 1 to 30.');
  }

  const dailyAttempts = MAX_ATTEMPTS * 3;
  const maxAttempts = difficulty >= 6 ? dailyAttempts : MAX_ATTEMPTS;

  for (let attempt = 0; attempt < maxAttempts; attempt += 1) {
    const random = seededRandom(seed + attempt * 0x9e3779b9);
    const level = difficulty >= 6
      ? createDailyLayout(seed, size, difficulty, random)
      : createLessonLayout(seed, size, difficulty, random);
    if (level) return level;
  }

  for (let attempt = 0; attempt < MAX_ATTEMPTS; attempt += 1) {
    const random = seededRandom(seed + attempt * 0x9e3779b9 + 0x12345);
    const level = createLessonLayout(seed, size, difficulty, random);
    if (level) return level;
  }
  return null;
};

  type,

  ) : (
    <svg
      className="component"

      fill="none"

    >
      <g strokeLinecap="butt" strokeLinejoin="round">

            ]

export function dailyLayout(now = new Date()) {
  const day = Math.floor(now.getTime() / 86400000);
  let seed = day >>> 0;
  const random = () => {
    seed = (Math.imul(seed, 1664525) + 1013904223) >>> 0;
    return seed / 4294967296;
  };
  const transform = (i: number) => {
    let x = i % 6,
      y = Math.floor(i / 6);
    if (day % 8 >= 4) x = 5 - x;
    for (let n = 0; n < day % 4; n++) {
      [x, y] = [5 - y, x];
    }
    return y * 6 + x;
  };
  const top = day % 3 === 0 ? 4 : 5,
    bottom = day % 5 < 2 ? 30 : 35;
  const upper = [12, 13, 14, 15, 9, 3, 4, ...(top === 5 ? [5] : [])];
  const lower = [
    15,
    21,
    27,
    ...(bottom === 30 ? [26, 25, 24, 30] : [28, 29, 35]),
  ];
  const used = new Set([...upper, ...lower]);
  const available = Array.from({ length: 36 }, (_, i) => i).filter(
    (i) => !used.has(i)
  );
  for (let i = available.length - 1; i > 0; i--) {
    const j = Math.floor(random() * (i + 1));
    [available[i], available[j]] = [available[j]!, available[i]!];
  }
  const fixed = [12, 15, 27, top, bottom];
  const hazards = available
    .filter((i) =>
      fixed.every(
        (j) =>
          Math.abs((i % 6) - (j % 6)) +
            Math.abs(Math.floor(i / 6) - Math.floor(j / 6)) >
          1
      )
    )
    .slice(0, 2);
  const level: DailyLevel = {
    size: 6,
    id: 31,
    title: 'Daily Circuit',
    mode: 'daily',
    date: now.toISOString().slice(0, 10),
    difficulty: 'STANDARD',
    source: transform(12),
    boosters: [15, 27].map(transform),
    targets: [
      { index: transform(top), voltage: top === 5 ? 1 : 2 },
      { index: transform(bottom), voltage: bottom === 30 ? 1 : 2 },
    ],
    walls: available
      .filter((i) => !hazards.includes(i))
      .slice(0, 3)
      .map(transform),
    hazards: hazards.map(transform),
    targetCost: 0,
  };
  const directions = (a: number, b: number) =>
    b === a - 6 ? 0 : b === a + 1 ? 1 : b === a + 6 ? 2 : 3;
  const solution = new Map<number, { type: number; rotation: number }>();
  for (const path of [upper, lower])
    for (let j = 1; j < path.length - 1; j++) {
      const i = path[j]!;
      if (fixed.includes(i)) continue;
      const cell = transform(i),
        a = directions(cell, transform(path[j - 1]!)),
        b = directions(cell, transform(path[j + 1]!));
      const straight = Math.abs(a - b) === 2;
      solution.set(cell, {
        type: straight ? 1 : 2,
        rotation: straight ? a % 2 : (a + 1) % 4 === b ? a : b,
      });
      level.targetCost += straight ? 10 : 15;
    }
  return { level, solution };
}

export type FixedType = 'none' | 'source' | 'target' | 'wall' | 'booster' | 'hazard';

export type Placement = {
  type: number;
  rotation: number;
  fixed: FixedType;
};

export type Puzzle = {
  size: number;
  source: number;
  targets: readonly { index: number; voltage: number }[];
  boosters: readonly number[];
  walls: readonly number[];
  hazards: readonly number[];
};

export type ReceiverResult = {
  index: number;
  voltage: number;
  requiredVoltage: number;
  met: boolean;
};

export type PuzzleEvaluation = {
  voltages: ReadonlyMap<number, number>;
  receivers: ReceiverResult[];
  hazardPowered: boolean;
  hardwareCost: number;
  totalCost: number;
  connections: ReadonlyMap<number, readonly number[]>;
};

type Direction = {
  row: number;
  column: number;
  direction: number;
  opposite: number;
};

export const DIRECTIONS: readonly Direction[] = [
  { row: -1, column: 0, direction: 0, opposite: 2 },
  { row: 0, column: 1, direction: 1, opposite: 3 },
  { row: 1, column: 0, direction: 2, opposite: 0 },
  { row: 0, column: -1, direction: 3, opposite: 1 },
];

export const names = ['Eraser', 'Conduit', 'Corner', 'Splitter'];
export const SOURCE_VOLTAGE = 5;
export const BOOSTER_VOLTAGE = 5;
const COMPONENT_COSTS: Record<number, number> = {
  1: 10,
  2: 15,
  3: 25,
};
const COMPONENT_ARMS: Record<number, readonly number[]> = {
  0: [],
  1: [0, 2],
  2: [0, 1],
  3: [0, 1, 2],
  4: [0, 1, 2, 3],
};
const ALL_ARMS = [0, 1, 2, 3];

const armsFor = (placement: Placement | undefined): number[] => {
  if (!placement || placement.fixed === 'wall') return [];
  if (placement.fixed !== 'none') return ALL_ARMS;
  const arms = COMPONENT_ARMS[placement.type] ?? [];
  return arms.map((arm) => (arm + placement.rotation) % 4);
};

const fixedTypeAt = (puzzle: Puzzle, index: number): FixedType => {
  if (index === puzzle.source) return 'source';
  if (puzzle.targets.some((target) => target.index === index)) return 'target';
  if (puzzle.boosters.includes(index)) return 'booster';
  if (puzzle.walls.includes(index)) return 'wall';
  if (puzzle.hazards.includes(index)) return 'hazard';
  return 'none';
};

export const evaluate = (
  puzzle: Puzzle,
  placements: readonly Placement[],
  additionalCost = 0
): PuzzleEvaluation => {
  const cellCount = puzzle.size * puzzle.size;
  const connections = new Map<number, readonly number[]>();
  const effectivePlacements: (Placement | undefined)[] = [];
  let hardwareCost = 0;

  for (let index = 0; index < cellCount; index += 1) {
    const placement = placements[index];
    if (!placement) {
      effectivePlacements.push(undefined);
      connections.set(index, []);
      continue;
    }
    const effectivePlacement = { ...placement, fixed: fixedTypeAt(puzzle, index) };
    effectivePlacements.push(effectivePlacement);
    connections.set(index, armsFor(effectivePlacement));
    if (effectivePlacement.fixed === 'none' && effectivePlacement.type > 0) {
      hardwareCost += COMPONENT_COSTS[effectivePlacement.type] ?? 10;
    }
  }

  const voltages = new Map<number, number>([[puzzle.source, SOURCE_VOLTAGE]]);
  const queue: { index: number; voltage: number }[] = [
    { index: puzzle.source, voltage: SOURCE_VOLTAGE },
  ];
  let hazardPowered = false;

  while (queue.length > 0) {
    queue.sort((left, right) => right.voltage - left.voltage);
    const current = queue.shift();
    if (!current || current.voltage <= 0) continue;

    const row = Math.floor(current.index / puzzle.size);
    const column = current.index % puzzle.size;
    const currentConnections = connections.get(current.index) ?? [];

    for (const direction of DIRECTIONS) {
      if (!currentConnections.includes(direction.direction)) continue;
      const nextRow = row + direction.row;
      const nextColumn = column + direction.column;
      if (
        nextRow < 0 ||
        nextRow >= puzzle.size ||
        nextColumn < 0 ||
        nextColumn >= puzzle.size
      ) {
        continue;
      }

      const nextIndex = nextRow * puzzle.size + nextColumn;
      const nextPlacement = effectivePlacements[nextIndex];
      const nextConnections = connections.get(nextIndex) ?? [];
      if (!nextPlacement || !nextConnections.includes(direction.opposite)) continue;
      if (nextPlacement.fixed === 'hazard') hazardPowered = true;

      const nextVoltage =
        nextPlacement.fixed === 'booster'
          ? BOOSTER_VOLTAGE
          : current.voltage - 1;
      if (
        nextVoltage > (voltages.get(nextIndex) ?? -1) &&
        nextVoltage >= 0
      ) {
        voltages.set(nextIndex, nextVoltage);
        queue.push({ index: nextIndex, voltage: nextVoltage });
      }
    }
  }

  const receivers = puzzle.targets.map((target) => {
    const voltage = voltages.get(target.index) ?? 0;
    return {
      index: target.index,
      voltage,
      requiredVoltage: target.voltage,
      met: voltage >= target.voltage,
    };
  });

  return {
    voltages,
    receivers,
    hazardPowered,
    hardwareCost,
    totalCost: hardwareCost + additionalCost,
    connections,
  };
};

import {
  DIRECTIONS,
  evaluate,
  BOOSTER_VOLTAGE,
  type Placement,
  type Puzzle,
  type PuzzleEvaluation,
} from './rules.ts';

type SolverResult = {
  placements: Placement[];
  cost: number;
};

type SearchContext = {
  deadline: number;
  best: SolverResult | null;
  stopped: boolean;
};

const SEARCH_TIME_LIMIT_MS = 2000;

export type SolverOptions = {
  timeLimitMs?: number;
  lockedEmptyIndices?: readonly number[];
};

const COMPONENT_OPTIONS = [1, 2, 3]
  .flatMap((type) =>
    Array.from({ length: 4 }, (_, rotation) => ({ type, rotation }))
  )
  .map(({ type, rotation }) => {
    const probePuzzle: Puzzle = {
      size: 2,
      source: 0,
      targets: [],
      boosters: [],
      walls: [],
      hazards: [],
    };
    const probePlacements: Placement[] = [
      { type: 4, rotation: 0, fixed: 'source' },
      { type, rotation, fixed: 'none' },
      { type: 0, rotation: 0, fixed: 'none' },
      { type: 0, rotation: 0, fixed: 'none' },
    ];
    const probe = evaluate(probePuzzle, probePlacements);
    return {
      type,
      rotation,
      arms: probe.connections.get(1) ?? [],
      cost: probe.hardwareCost,
    };
  })
  .sort((left, right) => left.cost - right.cost);

const createInitialPlacements = (
  puzzle: Puzzle,
  existingPlacements?: readonly Placement[]
): Placement[] =>
  Array.from({ length: puzzle.size * puzzle.size }, (_, index) => {
    const existing = existingPlacements?.[index];
    if (existing && existing.fixed === 'none' && existing.type > 0) {
      return { ...existing };
    }
    if (index === puzzle.source) return { type: 4, rotation: 0, fixed: 'source' };
    if (puzzle.targets.some((target) => target.index === index)) {
      return { type: 4, rotation: 0, fixed: 'target' };
    }
    if (puzzle.boosters.includes(index)) {
      return { type: 4, rotation: 0, fixed: 'booster' };
    }
    if (puzzle.walls.includes(index)) {
      return { type: 0, rotation: 0, fixed: 'wall' };
    }
    if (puzzle.hazards.includes(index)) {
      return { type: 4, rotation: 0, fixed: 'hazard' };
    }
    return { type: 0, rotation: 0, fixed: 'none' };
  });

const isBlocked = (
  puzzle: Puzzle,
  index: number,
  lockedEmptyIndices: ReadonlySet<number>
): boolean =>
  puzzle.walls.includes(index) ||
  puzzle.hazards.includes(index) ||
  lockedEmptyIndices.has(index);

const distance = (puzzle: Puzzle, left: number, right: number): number =>
  Math.abs(Math.floor(left / puzzle.size) - Math.floor(right / puzzle.size)) +
  Math.abs((left % puzzle.size) - (right % puzzle.size));

const canReachReceiver = (
  puzzle: Puzzle,
  evaluation: PuzzleEvaluation,
  receiverIndex: number,
  requiredVoltage: number
): boolean => {
  let bestRemainingVoltage = -1;
  for (const [index, voltage] of evaluation.voltages) {
    const optimisticVoltage = voltage - distance(puzzle, index, receiverIndex);
    bestRemainingVoltage = Math.max(bestRemainingVoltage, optimisticVoltage);
  }

  for (const boosterIndex of puzzle.boosters) {
    const optimisticVoltage =
      BOOSTER_VOLTAGE - distance(puzzle, boosterIndex, receiverIndex);
    bestRemainingVoltage = Math.max(bestRemainingVoltage, optimisticVoltage);
  }

  return bestRemainingVoltage >= requiredVoltage;
};

const canReachAllReceivers = (
  puzzle: Puzzle,
  evaluation: PuzzleEvaluation
): boolean =>
  evaluation.receivers
    .filter((receiver) => !receiver.met)
    .every((receiver) =>
      canReachReceiver(
        puzzle,
        evaluation,
        receiver.index,
        receiver.requiredVoltage
      )
    );

const isEvaluationValid = (
  evaluation: PuzzleEvaluation,
  protectedReceiverIndices: readonly number[]
): boolean =>
  !evaluation.hazardPowered &&
  protectedReceiverIndices.every((index) =>
    evaluation.receivers.some((receiver) => receiver.index === index && receiver.met)
  );

export const solve = (
  puzzle: Puzzle,
  existingPlacements?: readonly Placement[],
  options: SolverOptions = {}
): Placement[] | null => {
  const timeLimitMs = options.timeLimitMs ?? SEARCH_TIME_LIMIT_MS;
  const lockedEmptyIndices = new Set(options.lockedEmptyIndices ?? []);
  const context: SearchContext = {
    deadline: Date.now() + timeLimitMs,
    best: null,
    stopped: false,
  };
  const initialPlacements = createInitialPlacements(puzzle, existingPlacements);

  const connectRemainingReceivers = (placements: Placement[]): void => {
    if (context.stopped) return;
    if (Date.now() >= context.deadline) {
      context.stopped = true;
      return;
    }

    const evaluation = evaluate(puzzle, placements);
    if (evaluation.hazardPowered) return;
    if (context.best && evaluation.hardwareCost >= context.best.cost) return;
    if (!canReachAllReceivers(puzzle, evaluation)) return;

    const remainingReceivers = evaluation.receivers
      .filter((receiver) => !receiver.met)
      .sort((left, right) => distance(puzzle, puzzle.source, left.index) - distance(puzzle, puzzle.source, right.index));

    if (remainingReceivers.length === 0) {
      context.best = {
        placements: placements.map((placement) => ({ ...placement })),
        cost: evaluation.hardwareCost,
      };
      return;
    }

    const target = remainingReceivers[0];
    if (!target) return;
    const protectedReceiverIndices = evaluation.receivers
      .filter((receiver) => receiver.met)
      .map((receiver) => receiver.index);
    routeToReceiver(
      puzzle.source,
      null,
      target.index,
      target.requiredVoltage,
      placements,
      new Set([puzzle.source]),
      protectedReceiverIndices
    );
  };

  const routeToReceiver = (
    currentIndex: number,
    incomingArm: number | null,
    targetIndex: number,
    targetVoltage: number,
    placements: Placement[],
    path: Set<number>,
    protectedReceiverIndices: readonly number[]
  ): void => {
    if (context.stopped || Date.now() >= context.deadline) {
      context.stopped = true;
      return;
    }
    if (context.best && evaluate(puzzle, placements).hardwareCost >= context.best.cost) {
      return;
    }

    const currentRow = Math.floor(currentIndex / puzzle.size);
    const currentColumn = currentIndex % puzzle.size;
    const orderedDirections = [...DIRECTIONS].sort((left, right) => {
      const leftIndex = (currentRow + left.row) * puzzle.size + currentColumn + left.column;
      const rightIndex = (currentRow + right.row) * puzzle.size + currentColumn + right.column;
      return distance(puzzle, leftIndex, targetIndex) -
        distance(puzzle, rightIndex, targetIndex);
    });

    for (const direction of orderedDirections) {
      if (context.stopped) return;
      const nextRow = currentRow + direction.row;
      const nextColumn = currentColumn + direction.column;
      if (
        nextRow < 0 ||
        nextRow >= puzzle.size ||
        nextColumn < 0 ||
        nextColumn >= puzzle.size
      ) {
        continue;
      }

      const nextIndex = nextRow * puzzle.size + nextColumn;
      if (path.has(nextIndex) || isBlocked(puzzle, nextIndex, lockedEmptyIndices)) continue;

      const currentPlacement = placements[currentIndex];
      if (!currentPlacement) continue;
      const currentOptions = currentPlacement.fixed === 'none'
        ? existingPlacements?.[currentIndex]?.type
          ? [currentPlacement]
          : COMPONENT_OPTIONS
          .filter((component) =>
            component.arms.includes(direction.direction) &&
            (incomingArm === null || component.arms.includes(incomingArm))
          )
          .map((component) => ({
            type: component.type,
            rotation: component.rotation,
            fixed: 'none' as const,
          }))
        : [currentPlacement];

      for (const currentOption of currentOptions) {
        if (context.stopped) return;
        const nextPlacements = placements.slice();
        nextPlacements[currentIndex] = currentOption;
        const evaluation = evaluate(puzzle, nextPlacements);
        if (
          !isEvaluationValid(evaluation, protectedReceiverIndices) ||
          (context.best && evaluation.hardwareCost >= context.best.cost) ||
          !canReachAllReceivers(puzzle, evaluation)
        ) {
          continue;
        }
        const currentArms = evaluation.connections.get(currentIndex) ?? [];
        if (
          !currentArms.includes(direction.direction) ||
          (incomingArm !== null && !currentArms.includes(incomingArm))
        ) {
          continue;
        }

        const voltage = evaluation.voltages.get(currentIndex) ?? 0;
        if (voltage <= 0) continue;
        if (
          !canReachReceiver(
            puzzle,
            evaluation,
            targetIndex,
            targetVoltage
          )
        ) {
          continue;
        }

        if (
          nextIndex === targetIndex &&
          evaluation.receivers.some(
            (receiver) => receiver.index === targetIndex && receiver.met
          )
        ) {
          connectRemainingReceivers(nextPlacements);
          continue;
        }

        const nextPath = new Set(path);
        nextPath.add(nextIndex);
        routeToReceiver(
          nextIndex,
          direction.opposite,
          targetIndex,
          targetVoltage,
          nextPlacements,
          nextPath,
          protectedReceiverIndices
        );
      }
    }
  };

  connectRemainingReceivers(initialPlacements);
  return context.best?.placements ?? null;
};

let audioContext: AudioContext | undefined;
let muted = false;
let mutePreferenceLoaded = false;

const loadMutePreference = (): void => {
  if (mutePreferenceLoaded) return;
  mutePreferenceLoaded = true;

  try {
    if (typeof window !== 'undefined') {
      muted = window.sessionStorage.getItem('daily-circuit:sound-enabled') === 'false';
    }
  } catch (error) {
    console.error('Unable to restore sound setting:', error);
  }
};

export const isMuted = (): boolean => {
  loadMutePreference();
  return muted;
};

export const setMuted = (nextMuted: boolean): void => {
  muted = nextMuted;
  mutePreferenceLoaded = true;

  try {
    if (typeof window !== 'undefined') {
      window.sessionStorage.setItem('daily-circuit:sound-enabled', String(!nextMuted));
    }
  } catch (error) {
    console.error('Unable to save sound setting:', error);
  }
};

const getAudioContext = (): AudioContext | undefined => {
  if (typeof window === 'undefined' || typeof window.AudioContext !== 'function') {
    return undefined;
  }

  try {
    audioContext ??= new window.AudioContext();
    if (audioContext.state === 'suspended') {
      void audioContext.resume().catch((error: unknown) => {
        console.error('Unable to resume puzzle audio:', error);
      });
    }
    return audioContext;
  } catch (error) {
    console.error('Unable to initialize puzzle audio:', error);
    return undefined;
  }
};

const playTones = (tones: Tone[], peakGain: number): void => {
  if (isMuted()) return;

  try {
    const context = getAudioContext();
    if (!context) return;

    const startTime = context.currentTime;
    for (const tone of tones) {
      const oscillator = context.createOscillator();
      const gain = context.createGain();
      const start = startTime + (tone.startOffset ?? 0);
      const end = start + tone.duration;

      oscillator.type = tone.type;
      oscillator.frequency.setValueAtTime(tone.frequency, start);
      gain.gain.setValueAtTime(0.0001, start);
      gain.gain.exponentialRampToValueAtTime(peakGain, start + 0.012);
      gain.gain.exponentialRampToValueAtTime(0.0001, end);

      oscillator.connect(gain);
      gain.connect(context.destination);
      oscillator.start(start);
      oscillator.stop(end + 0.015);
    }
  } catch (error) {
    console.error('Unable to play puzzle sound:', error);
  }
};
